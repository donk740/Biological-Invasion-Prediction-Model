import shap
import xgboost as xgb
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold, train_test_split
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, confusion_matrix, roc_auc_score, roc_curve
import matplotlib.pyplot as plt
import seaborn as sns
import statistics


# 1. 加载数据集
data = pd.read_csv('path')
X = data[['Elevation','MTWM','MHWM','MTDM','MHDM','AMDTR','MTWS','MHWS','MAP','MTCS','MHCS','MAT']]
y = data['type']  # 标签

# 2. 定义模型参数
params = {
    'objective': 'binary:logistic',
    'max_depth': 8,
    'eta': 0.18,
    'subsample': 0.65,
    'colsample_bytree': 0.8,
    'eval_metric': 'logloss'
}
num_rounds = 500

# 3. 设置10折交叉验证
skf = StratifiedKFold(n_splits=10, shuffle=True, random_state=96)

roc_aucs = []

all_y_true = []
all_y_pred = []

fold = 1
for train_index, val_index in skf.split(X, y):
    print(f"\n========== Fold {fold} ==========")
    
    X_train_fold, X_val_fold = X.iloc[train_index], X.iloc[val_index]
    y_train_fold, y_val_fold = y.iloc[train_index], y.iloc[val_index]
    
    # 转换为DMatrix格式
    dtrain_fold = xgb.DMatrix(X_train_fold, y_train_fold)
    dval_fold = xgb.DMatrix(X_val_fold, y_val_fold)
    
    # 训练模型
    model_fold = xgb.train(params, dtrain_fold, num_rounds)
    
    # 预测验证集
    y_pred_proba_fold = model_fold.predict(dval_fold)
    y_pred_fold = (y_pred_proba_fold > 0.5).astype(int)
    
    # 计算指标
    roc_auc = roc_auc_score(y_val_fold, y_pred_proba_fold)
    
    # 存储
    roc_aucs.append(roc_auc)
    
    # 记录所有验证集的真实值和预测值
    all_y_true.extend(y_val_fold)
    all_y_pred.extend(y_pred_fold)
    
    # 打印当前fold的指标
    print(f"ROC AUC:   {roc_auc:.4f}")
    
    fold += 1

# 4. 计算所有fold的平均值和标准差
print("\n========== 10折交叉验证汇总 ==========")
print(f"平均ROC AUC: {np.mean(roc_aucs):.4f} (±{np.std(roc_aucs):.4f})")

# 5. 绘制整体混淆矩阵（基于所有验证集的预测合并）
conf_matrix_total = confusion_matrix(all_y_true, all_y_pred)
plt.figure(figsize=(6, 5))
sns.heatmap(conf_matrix_total, annot=True, fmt='d', cmap='Blues',
            xticklabels=['Predicted 0', 'Predicted 1'],
            yticklabels=['Actual 0', 'Actual 1'])
plt.title('Overall Confusion Matrix (10-fold CV)')
plt.xlabel('Predicted')
plt.ylabel('Actual')
plt.show()
